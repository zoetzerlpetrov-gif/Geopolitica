// Resultado de una fuente → respuesta simple «¿aparece?».
const APARECE = {
  verificado: ["si", "✅ Sí aparece"],
  encontrado: ["si", "✅ Sí aparece"],
  indicio: ["probable", "🟡 Probablemente sí"],
  sin_resultado: ["no", "❌ No aparece"],
  indeterminado: ["duda", "❔ No se pudo confirmar"],
  limitado: ["duda", "❔ No se pudo confirmar"]
};
const PESO = { si: 4, probable: 3, no: 2, duda: 1, sin_revisar: 0, sesion: 0 };
const aparece = (estado) => APARECE[estado] || APARECE.indeterminado;

// Redes principales: siempre aparecen en la tabla, en este orden. «alias» son los nombres con que llegan de las fuentes.
// «sesion»: la red no permite comprobarlo sin iniciar sesión (se respeta su política; no se raspa).
const REDES = [
  ["Facebook", ["facebook"], true], ["Instagram", ["instagram"], true], ["X (Twitter)", ["x", "twitter"]],
  ["TikTok", ["tiktok"]], ["YouTube", ["youtube"]], ["LinkedIn", ["linkedin"], true], ["Threads", ["threads"], true],
  ["Snapchat", ["snapchat"]], ["Telegram", ["telegram"]], ["Pinterest", ["pinterest"]], ["Reddit", ["reddit"], true],
  ["Tumblr", ["tumblr"]], ["Twitch", ["twitch"]], ["Bluesky", ["bluesky"]], ["Mastodon", ["mastodon"]],
  ["Flickr", ["flickr"]], ["SoundCloud", ["soundcloud"]], ["GitHub", ["github"]]
];

function element(parent, tag, text, className) {
  const node = document.createElement(tag);
  node.textContent = text;
  if (className) node.className = className;
  parent.append(node);
  return node;
}

const nombreRed = (source) => String(source || "").split(" · ")[0].trim();
const clave = (texto) => texto.toLowerCase().replace(/\s*\(.*\)$/, "").replace(/[^a-z0-9]/g, "");

/** Tabla de redes principales: una fila por red con Sí / No / No se pudo confirmar, sin enlaces. */
function filasRedes(direct, catalog, catalogoRevisado) {
  const porClave = new Map();
  const anotar = (k, fila) => {
    const previa = porClave.get(k);
    if (!previa || PESO[fila.tipo] > PESO[previa.tipo]) porClave.set(k, fila);
  };
  for (const r of [...direct.filter(x => x.kind === "alias"), ...catalog]) {
    const [tipo, texto] = aparece(r.status);
    anotar(clave(nombreRed(r.source)), { tipo, texto, fuente: r.kind === "catalogo" ? "catálogo público" : nombreRed(r.source) === "Wikidata" ? "Wikidata" : "consulta directa" });
    for (const ref of r.socialReferences || []) anotar(clave(ref.platform), { tipo: "si", texto: "✅ Sí aparece", fuente: "referencia pública en Wikidata" });
  }
  const ORDEN = { si: 0, probable: 1, no: 2, duda: 3, sesion: 4, sin_revisar: 5 }; // primero donde sí aparece
  return REDES.map(([nombre, alias, soloSesion]) => {
    const hallada = alias.map(a => porClave.get(clave(a))).filter(Boolean).sort((a, b) => PESO[b.tipo] - PESO[a.tipo])[0];
    if (hallada && (hallada.tipo !== "duda" || !soloSesion)) return { nombre, ...hallada };
    if (soloSesion) return { nombre, tipo: "sesion", texto: "🔒 No se puede saber sin iniciar sesión", fuente: "la red no publica perfiles sin sesión" };
    return hallada ? { nombre, ...hallada } : { nombre, tipo: "sin_revisar", texto: "➖ No revisada", fuente: catalogoRevisado ? "fuera de la categoría elegida" : "activa el catálogo ampliado" };
  }).sort((a, b) => ORDEN[a.tipo] - ORDEN[b.tipo]);
}

/** Otros sitios agrupados por categoría: solo los nombres donde aparece; los demás se cuentan. */
function gruposOtros(direct, catalog, info) {
  const principales = new Set([...REDES.flatMap(([, alias]) => alias), "wikidata"]); // Wikidata solo aporta referencias
  const grupos = new Map();
  const meter = (titulo, r) => {
    if (principales.has(clave(nombreRed(r.source)))) return;
    if (!grupos.has(titulo)) grupos.set(titulo, { si: [], probable: [], no: 0, duda: [] });
    const g = grupos.get(titulo);
    const [tipo] = aparece(r.status);
    if (tipo === "no") g.no += 1; else g[tipo].push(nombreRed(r.source));
  };
  for (const r of direct.filter(x => x.kind === "alias")) meter("Consultas directas (desarrollo, juegos, comunidades)", r);
  for (const r of catalog) {
    const s = (info?.sections || []).find(x => x.categories.includes(r.category));
    meter(s ? s.label : "Otros sitios públicos", r);
  }
  return [...grupos.entries()].map(([titulo, g]) => ({ titulo, ...g, si: [...new Set(g.si)].sort(), probable: [...new Set(g.probable)].sort(), duda: [...new Set(g.duda)].sort() }));
}

function tabla(parent, columnas, filas, clase = "tabla-huella") {
  const t = element(parent, "table", "", clase);
  const cab = element(element(t, "thead", ""), "tr", "");
  for (const c of columnas) element(cab, "th", c).scope = "col";
  const cuerpo = element(t, "tbody", "");
  for (const fila of filas) {
    const tr = element(cuerpo, "tr", "", fila.clase || "");
    fila.celdas.forEach((c, i) => {
      const td = element(tr, i === 0 ? "th" : "td", typeof c === "string" ? c : "");
      if (i === 0) td.scope = "row";
      if (typeof c === "function") c(td);
    });
  }
  return t;
}

function detalleCorreo(r) {
  const partes = [];
  if (r.detail) partes.push(r.detail);
  if (r.name) partes.push("Nombre visible: " + r.name);
  if (r.since && !Number.isNaN(Date.parse(r.since))) partes.push("Cuenta creada el " + new Date(r.since).toLocaleDateString("es-MX"));
  if (r.evidence?.length) partes.push(`${r.evidence.length} commit(s) público(s) con este correo`);
  if (r.profileLinks?.length) partes.push(`${r.profileLinks.length} perfil(es) de GitHub publican este correo`);
  if (r.declaredProfiles?.length) {
    const sitios = [...new Set(r.declaredProfiles.map(u => { try { return new URL(u).hostname.replace(/^www\./, ""); } catch { return ""; } }).filter(Boolean))];
    partes.push("Gravatar enlaza a: " + sitios.join(", "));
  }
  if (r.correlation) partes.push("Cruce de fuentes: " + r.correlation);
  return partes.join(" · ");
}

function detalleDominio(r) {
  const partes = [];
  if (r.detail) partes.push(r.detail);
  if (r.records?.length) partes.push(r.records.join(" | "));
  if (r.incidents?.length) {
    partes.push(r.incidents.map(i => [i.title, i.date, i.verified ? "verificado" : "sin verificar"].filter(Boolean).join(" ")).join("; "));
    if (r.count > r.incidents.length) partes.push(`(${r.incidents.length} de ${r.count} incidentes)`);
  }
  return partes.join(" · ") || "—";
}

function resultadoDominio(r) {
  const incidentes = /pwned|incidente/i.test(r.source);
  if (r.status === "sin_resultado") return incidentes ? "✅ Sin incidentes conocidos del sitio" : "❌ No configurado";
  if (r.status === "encontrado") return "⚠️ Hay incidentes publicados";
  if (r.status === "verificado") return "✅ Configurado";
  return "❔ No se pudo confirmar";
}

function botonAvatar(celda, r) {
  if (!r.avatar) return;
  const show = element(celda, "button", "Mostrar avatar", "secondary mini");
  show.type = "button";
  show.addEventListener("click", () => {
    const avatar = document.createElement("img");
    avatar.className = "avatar-publico";
    avatar.alt = "Avatar público publicado por " + r.source;
    avatar.referrerPolicy = "no-referrer";
    avatar.addEventListener("error", () => avatar.replaceWith(document.createTextNode("La fuente no pudo cargar el avatar.")), { once: true });
    avatar.src = r.avatar;
    show.replaceWith(avatar);
  }, { once: true });
}

function tarjetaResumen(parent, titulo, valor, nota, tipo) {
  const t = element(parent, "div", "", "resumen-tarjeta " + (tipo || ""));
  element(t, "span", titulo, "resumen-titulo");
  element(t, "b", valor);
  if (nota) element(t, "small", nota);
}

/** Dibuja todo el informe ordenado: resumen, redes principales, otros sitios, correo y dominio. */
function renderTodo() {
  if (!ultima) return;
  const { query, direct, catalog, catalogInfo, category, catalogDone } = ultima;
  const resumen = $("resumen"), redes = $("redes-principales"), otros = $("otros-sitios"), correo = $("res-correo"), dominio = $("res-dominio");
  for (const n of [resumen, redes, otros, correo, dominio]) n.replaceChildren();
  const correoRes = direct.filter(r => r.kind === "correo");
  const dominioRes = direct.filter(r => r.kind === "dominio");
  const filas = query.username ? filasRedes(direct, catalog, catalogDone) : [];
  const grupos = query.username ? gruposOtros(direct, catalog, catalogInfo) : [];

  // 1. Resumen
  const enRedes = filas.filter(f => f.tipo === "si" || f.tipo === "probable");
  const enOtros = grupos.reduce((n, g) => n + g.si.length + g.probable.length, 0);
  const correoSi = correoRes.filter(r => ["si"].includes(aparece(r.status)[0]));
  if (query.username) {
    tarjetaResumen(resumen, "Redes sociales principales", `${enRedes.length} de ${filas.length}`,
      enRedes.length ? enRedes.map(f => f.nombre).join(", ") : "No aparece en las que se pudieron revisar", enRedes.length ? "si" : "no");
    tarjetaResumen(resumen, "Otros sitios", String(enOtros), catalogDone ? "con perfil del alias" : catalog.length ? "catálogo en curso…" : "sin catálogo ampliado", enOtros ? "si" : "");
  }
  if (query.email) tarjetaResumen(resumen, "Correo", `${correoSi.length} de ${correoRes.length}`, "fuentes con coincidencia pública", correoSi.length ? "si" : "no");
  if (dominioRes.length) {
    const mx = dominioRes.find(r => r.source === "MX");
    tarjetaResumen(resumen, "Dominio " + query.domain, mx && aparece(mx.status)[0] === "si" ? "Recibe correo" : "Sin correo (MX)", "", "");
  }

  // 2. Redes principales
  let n = 0;
  if (query.username) {
    element(redes, "h3", `${++n}. Redes sociales principales · @${query.username}`);
    element(redes, "p", "Solo indica si el alias aparece. Un alias igual puede pertenecer a otra persona: confírmalo antes de atribuirlo.", "nota");
    tabla(redes, ["Red social", "¿Aparece?", "Cómo se supo"], filas.map(f => ({ clase: "fila-" + f.tipo, celdas: [f.nombre, f.texto, f.fuente] })));
  }

  // 3. Otros sitios por categoría
  if (grupos.length) {
    element(otros, "h3", `${++n}. Otros sitios por categoría`);
    tabla(otros, ["Categoría", "✅ Aparece", "🟡 Probable", "❌ No aparece", "❔ Sin confirmar"], grupos.map(g => ({ celdas: [
      g.titulo, g.si.join(", ") || "—", g.probable.join(", ") || "—", String(g.no),
      (td) => {
        if (!g.duda.length) { td.textContent = "0"; return; }
        const d = element(td, "details", "");
        element(d, "summary", String(g.duda.length));
        element(d, "p", g.duda.join(", "));
      }] })));
  }

  // 4. Correo
  if (correoRes.length) {
    element(correo, "h3", `${++n}. Correo ${query.email}`);
    tabla(correo, ["Fuente", "Resultado", "Detalle"], correoRes.map(r => ({ clase: "fila-" + aparece(r.status)[0], celdas: [
      r.source, aparece(r.status)[1].replace("aparece", "hay coincidencia"), (td) => { td.textContent = detalleCorreo(r) || "—"; botonAvatar(td, r); }] })));
    if (!query.username) renderEmailPivots(direct, correo);
  }

  // 5. Dominio
  if (dominioRes.length) {
    element(dominio, "h3", `${++n}. Dominio ${query.domain}`);
    tabla(dominio, ["Comprobación", "Resultado", "Detalle"], dominioRes.map(r => ({ clase: "fila-" + aparece(r.status)[0], celdas: [
      r.source, resultadoDominio(r), detalleDominio(r)] })));
  }
  $("exportar").hidden = false;
}

/** CSV de todo el informe (sin enlaces): Sección, Fuente, Resultado, Detalle. */
function csvInforme() {
  const { query, direct, catalog, catalogInfo, catalogDone } = ultima;
  const filas = [["Sección", "Fuente", "Resultado", "Detalle"]];
  if (query.username) {
    for (const f of filasRedes(direct, catalog, catalogDone)) filas.push(["Redes principales", f.nombre, f.texto.replace(/^\S+\s/, ""), f.fuente]);
    for (const g of gruposOtros(direct, catalog, catalogInfo)) {
      for (const n of g.si) filas.push([g.titulo, n, "Sí aparece", ""]);
      for (const n of g.probable) filas.push([g.titulo, n, "Probablemente sí", ""]);
    }
  }
  for (const r of direct.filter(x => x.kind === "correo")) filas.push(["Correo", r.source, aparece(r.status)[1].replace(/^\S+\s/, ""), detalleCorreo(r)]);
  for (const r of direct.filter(x => x.kind === "dominio")) filas.push(["Dominio", r.source, resultadoDominio(r).replace(/^\S+\s/, ""), detalleDominio(r)]);
  const celda = (v) => { let s = String(v ?? ""); if (/^[=+\-@]/.test(s)) s = "'" + s; return /[",\n;]/.test(s) ? `"${s.replace(/"/g, '""')}"` : s; };
  return "﻿" + filas.map(f => f.map(celda).join(",")).join("\r\n");
}

async function runCatalog(username, info, category, signal) {
  const section = info.sections?.find(entry => entry.id === category);
  const total = section?.total ?? info.total;
  const batchSize = info.batchSize;
  const resultsByOffset = new Map();
  let nextOffset = 0;
  let completed = 0;
  let expired = false;
  let ultimoDibujo = 0;
  const run = async () => {
    while (!signal.aborted && !expired && nextOffset < total) {
      const offset = nextOffset;
      nextOffset += batchSize;
      try {
        const response = await fetch("consulta/catalogo", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ username, offset, category }),
          signal,
          redirect: "error",
          cache: "no-store"
        });
        if (response.status === 403) expired = true;
        if (response.ok) {
          const batch = await response.json();
          if (!signal.aborted && batch.offset === offset && Array.isArray(batch.results)) {
            resultsByOffset.set(offset, batch.results);
            completed += batch.results.length;
          }
        }
      } catch {
        // Continúa con los demás lotes si la conexión de uno falla.
      }
      if (signal.aborted) return;
      status.textContent = `Catálogo ampliado · ${completed} de ${total} fuentes revisadas…`;
      if (Date.now() - ultimoDibujo > 4000) { // el informe se va llenando mientras avanza
        ultimoDibujo = Date.now();
        ultima.catalog = [...resultsByOffset.keys()].sort((a, b) => a - b).flatMap(o => resultsByOffset.get(o));
        renderTodo();
      }
    }
  };
  await Promise.all([run(), run()]);
  if (signal.aborted) return;
  ultima.catalog = [...resultsByOffset.keys()].sort((a, b) => a - b).flatMap(o => resultsByOffset.get(o));
  ultima.catalogDone = true;
  renderTodo();
  status.textContent = expired ? "La sesión venció durante el catálogo. Los resultados completados siguen visibles." :
    `Listo · ${completed} de ${total} fuentes del catálogo revisadas.` + (completed < total ? ` Quedaron ${total - completed} sin revisar.` : "");
}

const PIVOT_SITES = {
  "github.com": ["GitHub", /^\/([a-z0-9-]+)$/i],
  "bsky.app": ["Bluesky", /^\/profile\/([a-z0-9.-]+)$/i],
  "x.com": ["X", /^\/([a-z0-9_]+)$/i],
  "instagram.com": ["Instagram", /^\/([a-z0-9_.]+)$/i],
  "www.instagram.com": ["Instagram", /^\/([a-z0-9_.]+)$/i],
  "www.tiktok.com": ["TikTok", /^\/@([a-z0-9_.]+)$/i],
  "www.linkedin.com": ["LinkedIn", /^\/in\/([a-z0-9-]+)$/i],
  "www.youtube.com": ["YouTube", /^\/@([a-z0-9_.-]+)$/i],
  "www.reddit.com": ["Reddit", /^\/(?:user|u)\/([a-z0-9_-]+)$/i],
  "medium.com": ["Medium", /^\/@([a-z0-9_.-]+)$/i],
  "t.me": ["Telegram", /^\/([a-z0-9_]+)$/i]
};

/** Alias que el propio correo publica (Gravatar, GitHub): botones para buscarlos en redes con un clic. */
function renderEmailPivots(entries, parent) {
  const links = entries.flatMap(result => {
    if (result.kind !== "correo" || result.status !== "verificado") return [];
    if (result.source === "GitHub · correo del perfil público") return result.profileLinks || [];
    return result.declaredProfiles || [];
  });
  const pivots = new Map();
  for (const link of links) {
    let url;
    try { url = new URL(link); } catch { continue; }
    if (!Object.hasOwn(PIVOT_SITES, url.hostname)) continue;
    const [platform, pattern] = PIVOT_SITES[url.hostname];
    const handle = pattern.exec(url.pathname.replace(/\/$/, ""))?.[1];
    if (handle && /^[a-z0-9](?:[a-z0-9_.-]{0,63}[a-z0-9_-])?$/i.test(handle) && !pivots.has(handle.toLowerCase())) {
      pivots.set(handle.toLowerCase(), { platform, handle });
    }
  }
  if (!pivots.size) return;
  element(parent, "p", "Siguiente paso: el correo publica estos alias. Elige uno para ver en qué redes aparece:");
  const actions = element(parent, "div", "", "actions");
  for (const { platform, handle } of pivots.values()) {
    const button = element(actions, "button", `Buscar @${handle} (${platform})`, "secondary");
    button.type = "button";
    button.addEventListener("click", () => {
      $("alias").value = handle;
      updateScope();
      $("alias").focus();
    });
  }
}

function updateScope() {
  const input = $("correo").value.trim();
  const hasEmail = input.includes("@");
  const hasAlias = Boolean($("alias").value.trim().replace(/^@/, ""));
  const catalog = $("catalogo-ampliado");
  $("opciones-correo").hidden = !hasEmail;
  catalog.disabled = !hasAlias;
  $("catalogo-categoria").disabled = !hasAlias || !catalog.checked;
  const scope = [];
  if (hasEmail && $("modo-correo").value === "exacto") scope.push("Se harán 6 comprobaciones del correo y 4 del dominio.");
  else if (input) scope.push("Se harán 4 comprobaciones del dominio.");
  const selection = $("catalogo-categoria");
  if (hasAlias) scope.push(catalog.checked
    ? `Se revisarán las redes principales y el catálogo: ${selection.selectedOptions?.[0]?.textContent?.trim() || "categoría seleccionada"}.`
    : "Se revisarán solo las fuentes directas por alias.");
  else if (input) scope.push("Para saber en qué redes aparece, escribe también un alias público.");
  $("alcance-consulta").textContent = scope.join(" ") || "Escribe un correo, un dominio o un alias público.";
}

function limpiar() {
  ultima = null;
  for (const id of ["resumen", "redes-principales", "otros-sitios", "res-correo", "res-dominio"]) $(id).replaceChildren();
  $("exportar").hidden = true;
  $("hallazgos").hidden = true;
}

form.addEventListener("input", updateScope);
form.addEventListener("change", updateScope);
updateScope();

form.addEventListener("submit", async (event) => {
  event.preventDefault();
  if (pending) return;
  const input = $("correo").value.trim();
  const username = $("alias").value.trim().replace(/^@/, "");
  const [local, domain] = input.includes("@") ? input.split("@") : ["", input];
  if (input.includes("@") && (!local || !domain || input.indexOf("@") !== input.lastIndexOf("@"))) {
    status.textContent = "Revisa el formato del correo.";
    return;
  }
  const expand = $("catalogo-ampliado").checked && Boolean(username);
  const category = $("catalogo-categoria").value;
  const query = { domain, username };
  if (input.includes("@") && $("modo-correo").value === "exacto") {
    if (!local || !/^[a-z0-9_%+-]+(?:\.[a-z0-9_%+-]+)*$/i.test(local) || local.length > 64) {
      status.textContent = "Para consultar el correo exacto, escribe una dirección compatible, como equipo@example.org.";
      return;
    }
    Object.assign(query, { email: input, searchEmail: true });
  }
  const current = new AbortController();
  pending = current;
  search.disabled = true;
  limpiar();
  status.textContent = "Consultando fuentes públicas…";
  try {
    const response = await fetch("consulta", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(query),
      signal: current.signal,
      redirect: "error",
      cache: "no-store"
    });
    if (!response.ok) {
      status.textContent = response.status === 400 ? "Revisa el formato del dominio, correo o alias." : "La consulta no está disponible. Comprueba tu sesión de administrador.";
      return;
    }
    const data = await response.json();
    if (current.signal.aborted) return;
    ultima = { query, direct: data.results, catalog: [], catalogInfo: data.catalog, category, catalogDone: !expand };
    renderTodo();
    $("marca-tiempo").textContent = "Consulta del " + new Date(data.checkedAt).toLocaleString("es-MX");
    $("hallazgos").hidden = false;
    status.textContent = expand && data.catalog ? "Fuentes directas listas. Revisando el catálogo ampliado…" : `Listo · ${data.results.length} comprobaciones.`;
    if (expand && data.catalog) await runCatalog(username, data.catalog, category, current.signal);
  } catch {
    if (!current.signal.aborted) status.textContent = "No se pudo completar la consulta. Comprueba la conexión y tu sesión.";
  } finally {
    if (pending === current) {
      pending = null;
      search.disabled = false;
    }
  }
});

$("exportar").addEventListener("click", () => {
  if (!ultima) return;
  const a = document.createElement("a");
  a.href = URL.createObjectURL(new Blob([csvInforme()], { type: "text/csv;charset=utf-8" }));
  a.download = `huella-${new Date().toISOString().slice(0, 10)}.csv`;
  a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 1000);
});

form.addEventListener("reset", () => {
  pending?.abort();
  pending = null;
  search.disabled = false;
  limpiar();
  status.textContent = "Consulta y resultados borrados de la pantalla.";
  queueMicrotask(updateScope);
});
search.disabled = false;

$("salir").addEventListener("click", async () => {
  try {
    const response = await fetch("salir", { method: "POST", redirect: "error" });
    if (![204, 403].includes(response.status)) throw new Error();
    location.replace("ingreso");
  } catch {
    status.textContent = "No se pudo cerrar la sesión. Inténtalo de nuevo.";
  }
});
