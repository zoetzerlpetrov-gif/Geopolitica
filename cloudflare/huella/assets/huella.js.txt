function element(parent, tag, text, className) {
  const node = document.createElement(tag);
  node.textContent = text;
  if (className) node.className = className;
  parent.append(node);
  return node;
}

function render(group, result) {
  const card = element(group, "article", "", "panel");
  element(card, "h3", result.source);
  element(card, "p", labels[result.status], "state " + result.status);
  if (result.detail) element(card, "p", result.detail);
  if (result.correlation) element(card, "p", "Cruce de fuentes · " + result.correlation, "state verificado");
  if (result.name) element(card, "p", result.name);
  if (result.avatar) {
    const show = element(card, "button", "Mostrar avatar público", "secondary");
    show.type = "button";
    show.addEventListener("click", () => {
      const avatar = document.createElement("img");
      avatar.className = "avatar-publico";
      avatar.alt = "Avatar público publicado por " + result.source;
      avatar.referrerPolicy = "no-referrer";
      avatar.addEventListener("error", () => {
        avatar.replaceWith(document.createTextNode("La fuente no pudo cargar el avatar."));
      }, { once: true });
      avatar.src = result.avatar;
      show.replaceWith(avatar);
    }, { once: true });
  }
  if (result.since && !Number.isNaN(Date.parse(result.since))) {
    element(card, "p", "Cuenta creada el " + new Date(result.since).toLocaleDateString("es-MX"));
  }
  if (result.records?.length) {
    const list = element(card, "ul", "", "records");
    for (const record of result.records) element(list, "li", record);
  }
  if (result.incidents?.length) {
    const list = element(card, "ul", "", "records");
    for (const incident of result.incidents) {
      const state = incident.verified ? "verificado por HIBP" : "sin verificar por HIBP";
      const entry = element(list, "li", [incident.title, incident.date, state].filter(Boolean).join(" · "));
      if (incident.classes.length) element(entry, "small", incident.classes.join(", "));
    }
    if (result.count > result.incidents.length) element(card, "p", `Mostrando ${result.incidents.length} de ${result.count} incidentes.`);
  }
  if (result.declaredProfiles?.length) {
    element(card, "p", "Enlaces visibles publicados en Gravatar");
    const list = element(card, "ul", "", "records");
    for (const url of result.declaredProfiles) {
      const link = element(element(list, "li", ""), "a", url);
      link.href = url;
      link.target = "_blank";
      link.rel = "noopener noreferrer";
    }
  }
  if (result.socialReferences?.length) {
    const list = element(card, "ul", "", "records");
    for (const entry of result.socialReferences) {
      const row = element(list, "li", "");
      for (const [url, label] of [[entry.url, entry.platform], [entry.reference, " · ficha en Wikidata"]]) {
        const link = element(row, "a", label);
        link.href = url;
        link.target = "_blank";
        link.rel = "noopener noreferrer";
      }
    }
  }
  if (result.profileLinks?.length) {
    const list = element(card, "ul", "", "records");
    for (const url of result.profileLinks) {
      const link = element(element(list, "li", ""), "a", "Abrir perfil coincidente");
      link.href = url;
      link.target = "_blank";
      link.rel = "noopener noreferrer";
    }
  }
  if (result.evidence?.length) {
    const list = element(card, "ul", "", "records");
    for (const url of result.evidence) {
      const link = element(element(list, "li", ""), "a", "Abrir commit público");
      link.href = url;
      link.target = "_blank";
      link.rel = "noopener noreferrer";
    }
  }
  if (result.url) {
    const link = element(card, "a", result.kind === "catalogo" ? "Revisar página en la fuente" : "Abrir perfil público");
    link.href = result.url;
    link.target = "_blank";
    link.rel = "noopener noreferrer";
  }
}

function renderManualNetworks(username) {
  const alias = username.replace(/^@/, "");
  if (!alias) return;
  if (/^[a-z0-9_.]{1,30}$/i.test(alias)) {
    const card = element(manualResults, "section", "", "panel");
    element(card, "h3", "Instagram · revisión manual");
    element(card, "p", "El enlace abre el perfil en Instagram. Las referencias públicas, si existen, aparecen en la tarjeta de Wikidata.");
    const link = element(card, "a", `Abrir @${alias} en Instagram`);
    link.href = `https://www.instagram.com/${encodeURIComponent(alias)}/`;
    link.target = "_blank";
    link.rel = "noopener noreferrer";
  }
  const details = element(manualResults, "details", "");
  element(details, "summary", `Redes y sitios populares · ${manualNetworks.length} enlaces para revisar`);
  element(details, "p", "Enlaces sugeridos a partir del alias. Huella no visitó estas direcciones en esta sección. Algunos sitios requieren iniciar sesión o usan otro identificador.");
  const list = element(details, "ul", "", "redes-lista");
  for (const [name, template] of manualNetworks) {
    const link = element(element(list, "li", ""), "a", name);
    link.href = template.replace("{}", encodeURIComponent(alias));
    link.target = "_blank";
    link.rel = "noopener noreferrer";
  }
}

function renderGroup(parent, title, entries) {
  if (!entries.length) return;
  const matched = entries.filter(result => ["verificado", "encontrado", "indicio"].includes(result.status));
  const other = entries.filter(result => !matched.includes(result));
  const section = element(parent, "section", "");
  element(section, "h3", `${title} · ${matched.length} de ${entries.length} con resultado`);
  if (matched.length) {
    const group = element(section, "div", "", "grid");
    for (const result of matched) render(group, result);
  }
  if (other.length) {
    const details = element(section, "details", "");
    const unresolved = other.filter(result => result.status === "indeterminado" || result.status === "limitado").length;
    element(details, "summary", `Ver las otras ${other.length} comprobaciones` +
      (unresolved ? ` · ${unresolved} sin respuesta concluyente` : ""));
    const group = element(details, "div", "", "grid");
    for (const result of other) render(group, result);
  }
}

function renderCoverage(parent, sections, entries) {
  if (!sections.length) return;
  const table = element(parent, "table", "", "cobertura-catalogo");
  element(table, "caption", "Cobertura del catálogo por categoría");
  const heading = element(element(table, "thead", ""), "tr", "");
  for (const text of ["Categoría", "Procesadas", "Con indicio", "No concluyentes"]) {
    element(heading, "th", text).scope = "col";
  }
  const rows = element(table, "tbody", "");
  for (const section of sections) {
    const matches = entries.filter(result => section.categories.includes(result.category));
    const row = element(rows, "tr", "");
    element(row, "th", section.label).scope = "row";
    element(row, "td", `${matches.length} de ${section.total}`);
    element(row, "td", String(matches.filter(result => result.status === "indicio").length));
    element(row, "td", String(matches.filter(result => ["indeterminado", "limitado"].includes(result.status)).length));
  }
}

async function runCatalog(username, info, category, signal, directResults) {
  const section = info.sections?.find(entry => entry.id === category);
  const total = section?.total ?? info.total;
  const batchSize = info.batchSize;
  const resultsByOffset = new Map();
  const profileLinks = new Set(directResults.flatMap(result => result.declaredProfiles || []));
  let nextOffset = 0;
  let completed = 0;
  let expired = false;
  const run = async () => {
    while (!signal.aborted && !expired && nextOffset < total) {
      const offset = nextOffset;
      nextOffset += batchSize;
      try {
        const response = await fetch("consulta/catalogo", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ username, offset, category }),
          signal,
          redirect: "error",
          cache: "no-store"
        });
        if (response.status === 403) expired = true;
        if (response.ok) {
          const batch = await response.json();
          if (!signal.aborted && batch.offset === offset && Array.isArray(batch.results)) {
            resultsByOffset.set(offset, batch.results);
            completed += batch.results.length;
          }
        }
      } catch {
        // Continúa con los demás lotes si la conexión de uno falla.
      }
      if (!signal.aborted) status.textContent = `Catálogo ampliado · ${completed} de ${total} fuentes procesadas.`;
    }
  };
  await Promise.all([run(), run()]);
  if (signal.aborted) return;
  const entries = [...resultsByOffset.keys()].sort((a, b) => a - b).flatMap(offset => resultsByOffset.get(offset));
  for (const result of entries) {
    if (result.status === "indicio" && profileLinks.has(result.url.replace(/\/$/, ""))) {
      result.correlation = "Gravatar publica un enlace visible a esta misma página.";
    }
  }
  renderCoverage(catalogResults, section ? [section] : (info.sections || []), entries);
  renderGroup(catalogResults, section ? `Catálogo ampliado · ${section.label}` : "Catálogo ampliado", entries);
  status.textContent = expired ? "La sesión venció durante el catálogo. Los resultados completados siguen visibles." :
    `Catálogo ampliado · ${completed} de ${total} fuentes procesadas.` +
    (completed < total ? ` Quedaron ${total - completed} sin procesar.` : "");
}

const PIVOT_SITES = {
  "github.com": ["GitHub", /^\/([a-z0-9-]+)$/i],
  "bsky.app": ["Bluesky", /^\/profile\/([a-z0-9.-]+)$/i],
  "x.com": ["X", /^\/([a-z0-9_]+)$/i],
  "instagram.com": ["Instagram", /^\/([a-z0-9_.]+)$/i],
  "www.instagram.com": ["Instagram", /^\/([a-z0-9_.]+)$/i],
  "www.tiktok.com": ["TikTok", /^\/@([a-z0-9_.]+)$/i],
  "www.linkedin.com": ["LinkedIn", /^\/in\/([a-z0-9-]+)$/i],
  "www.youtube.com": ["YouTube", /^\/@([a-z0-9_.-]+)$/i],
  "www.reddit.com": ["Reddit", /^\/(?:user|u)\/([a-z0-9_-]+)$/i],
  "medium.com": ["Medium", /^\/@([a-z0-9_.-]+)$/i],
  "t.me": ["Telegram", /^\/([a-z0-9_]+)$/i]
};

function renderEmailPivots(entries) {
  const links = entries.flatMap(result => {
    if (result.kind !== "correo" || result.status !== "verificado") return [];
    if (result.source === "GitHub · correo del perfil público") return result.profileLinks || [];
    return result.declaredProfiles || [];
  });
  const pivots = new Map();
  for (const link of links) {
    const url = new URL(link);
    if (!Object.hasOwn(PIVOT_SITES, url.hostname)) continue;
    const [platform, pattern] = PIVOT_SITES[url.hostname];
    const handle = pattern.exec(url.pathname.replace(/\/$/, ""))?.[1];
    if (handle && /^[a-z0-9](?:[a-z0-9_.-]{0,63}[a-z0-9_-])?$/i.test(handle) && !pivots.has(handle.toLowerCase())) {
      pivots.set(handle.toLowerCase(), { platform, handle });
    }
  }
  if (!pivots.size) return;
  element(manualResults, "p", "Alias publicados que puedes consultar por separado:");
  const actions = element(manualResults, "div", "", "actions");
  for (const { platform, handle } of pivots.values()) {
    const button = element(actions, "button", `Usar @${handle} · ${platform}`, "secondary");
    button.type = "button";
    button.addEventListener("click", () => {
      $("alias").value = handle;
      updateScope();
      $("alias").focus();
    });
  }
}

function updateScope() {
  const input = $("correo").value.trim();
  const hasEmail = input.includes("@");
  const hasAlias = Boolean($("alias").value.trim().replace(/^@/, ""));
  const catalog = $("catalogo-ampliado");
  $("opciones-correo").hidden = !hasEmail;
  catalog.disabled = !hasAlias;
  $("catalogo-categoria").disabled = !hasAlias || !catalog.checked;
  const scope = [];
  if (hasEmail && $("modo-correo").value === "exacto") scope.push("Se harán 6 comprobaciones por correo exacto y 4 de dominio.");
  else if (input) scope.push("Se harán 4 comprobaciones del dominio.");
  const selection = $("catalogo-categoria");
  if (hasAlias) scope.push(catalog.checked
    ? `Se consultarán las fuentes directas por alias y el catálogo: ${selection.selectedOptions?.[0]?.textContent?.trim() || "categoría seleccionada"}.`
    : "Se consultarán las fuentes directas por alias.");
  else if (input) scope.push("El catálogo de redes, comunidades y blogs requiere un alias público.");
  $("alcance-consulta").textContent = scope.join(" ") || "Escribe un correo, un dominio o un alias público.";
}

function describeEmailScope(query, entries) {
  if (query.email) {
    const emailResults = entries.filter(result => result.kind === "correo");
    const hits = emailResults.filter(result => result.status === "verificado" || result.status === "encontrado").length;
    const unresolved = emailResults.filter(result => result.status === "indeterminado" || result.status === "limitado").length;
    element(results, "p", `Correo exacto · ${hits} de ${emailResults.length} comprobaciones con coincidencia pública.` +
      (unresolved ? ` ${unresolved} quedaron sin una respuesta concluyente.` : ""));
  }
  if (!query.username && query.domain) {
    element(results, "p", "Para buscar en redes, comunidades y blogs, escribe un alias público, como el identificador que aparece después de @ en el perfil.");
  }
}

form.addEventListener("input", updateScope);
form.addEventListener("change", updateScope);
updateScope();

form.addEventListener("submit", async (event) => {
  event.preventDefault();
  if (pending) return;
  const input = $("correo").value.trim();
  const username = $("alias").value.trim().replace(/^@/, "");
  const [local, domain] = input.includes("@") ? input.split("@") : ["", input];
  if (input.includes("@") && (!local || !domain || input.indexOf("@") !== input.lastIndexOf("@"))) {
    status.textContent = "Revisa el formato del correo.";
    return;
  }
  const expand = $("catalogo-ampliado").checked && Boolean(username);
  const category = $("catalogo-categoria").value;
  const query = { domain, username };
  if (input.includes("@") && $("modo-correo").value === "exacto") {
    if (!local || !/^[a-z0-9_%+-]+(?:\.[a-z0-9_%+-]+)*$/i.test(local) || local.length > 64) {
      status.textContent = "Para consultar el correo exacto, escribe una dirección compatible, como equipo@example.org.";
      return;
    }
    Object.assign(query, { email: input, searchEmail: true });
  }
  const current = new AbortController();
  pending = current;
  search.disabled = true;
  $("hallazgos").hidden = true;
  results.replaceChildren();
  catalogResults.replaceChildren();
  manualResults.replaceChildren();
  status.textContent = "Consultando fuentes públicas…";
  try {
    const response = await fetch("consulta", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(query),
      signal: current.signal,
      redirect: "error",
      cache: "no-store"
    });
    if (!response.ok) {
      status.textContent = response.status === 400 ? "Revisa el formato del dominio, correo o alias." : "La consulta no está disponible. Comprueba tu sesión de administrador.";
      return;
    }
    const data = await response.json();
    if (current.signal.aborted) return;
    describeEmailScope(query, data.results);
    renderGroup(results, "Correo exacto", data.results.filter(result => result.kind === "correo"));
    const aliasResults = data.results.filter(result => result.kind === "alias");
    renderGroup(results, "Redes, comunidades y blogs · consulta directa", aliasResults.filter(result => result.category === "social"));
    renderGroup(results, "Otras consultas directas por alias", aliasResults.filter(result => result.category !== "social"));
    renderGroup(results, "Dominio", data.results.filter(result => result.kind === "dominio"));
    renderManualNetworks(username);
    if (!username && query.email) renderEmailPivots(data.results);
    $("marca-tiempo").textContent = "Consulta del " + new Date(data.checkedAt).toLocaleString("es-MX");
    $("hallazgos").hidden = false;
    status.textContent = `${data.results.length} comprobaciones directas terminadas.`;
    if (expand && data.catalog) await runCatalog(username, data.catalog, category, current.signal, data.results);
  } catch {
    if (!current.signal.aborted) status.textContent = "No se pudo completar la consulta. Comprueba la conexión y tu sesión.";
  } finally {
    if (pending === current) {
      pending = null;
      search.disabled = false;
    }
  }
});

form.addEventListener("reset", () => {
  pending?.abort();
  pending = null;
  search.disabled = false;
  $("hallazgos").hidden = true;
  results.replaceChildren();
  catalogResults.replaceChildren();
  manualResults.replaceChildren();
  status.textContent = "Consulta y resultados borrados de la pantalla.";
  queueMicrotask(updateScope);
});
search.disabled = false;

$("salir").addEventListener("click", async () => {
  try {
    const response = await fetch("salir", { method: "POST", redirect: "error" });
    if (![204, 403].includes(response.status)) throw new Error();
    location.replace("ingreso");
  } catch {
    status.textContent = "No se pudo cerrar la sesión. Inténtalo de nuevo.";
  }
});
